# 环境安装

编辑器 VSCode

## 系统

WSL2 Ubuntu

## make 工具安装

```bash
apt install make
```

## python 环境安装

自带的一个 python3.12，但是没有 pip 之类的

```bash
sudo apt install python3-pip python3-venv
```

## 修改一下 Makefile 的内容

```bash
.PHONY: help
.SILENT:

help:
	@grep -E '^[a-zA-Z_-]+:.*?# .*$$' $(MAKEFILE_LIST) | sort | awk 'BEGIN {FS = ":.*?# "}; {printf "\033[36m%-30s\033[0m %s\n", $$1, $$2}'

install-python-requirements:  # Install Python 3 required libraries
	@echo "创建虚拟环境..."
	python3 -m venv venv
	@echo "安装依赖..."
	./venv/bin/pip install --upgrade pip
	./venv/bin/pip install -r requirements.txt
	@echo "✅ 完成！使用 'source venv/bin/activate' 激活环境"

generate-site: install-python-requirements  # Use custom-script to generate the website
	@echo "生成网站..."
	cd scripts && bash Generate_Site_mkDocs.sh

serve:  # Start's a Python http.server on port 8000 serving the content of ./generated/site
	@echo "启动服务器在 http://localhost:8000"
	python3 -m http.server -d generated/site

clean:  # Clean up ephemeral build directories from the repo
	rm -rf generated venv
```

依赖安装：

```bash
make install-python-requirements
```

生成网站静态代码

```bash
make generate-site
```

本地起服务，预览网站

```bash
make serve
```
